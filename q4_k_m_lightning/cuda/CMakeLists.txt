cmake_minimum_required(VERSION 3.24)
project(lightning_q4_ops LANGUAGES CXX CUDA)
set(ORT_INCLUDE_DIR "" CACHE PATH "ONNX Runtime session headers")
set(BF16_SOURCE_DIR "" CACHE PATH "Qualified bf16_cuda_lightning source directory")
if(NOT EXISTS "${ORT_INCLUDE_DIR}/onnxruntime_cxx_api.h")
  message(FATAL_ERROR "Set ORT_INCLUDE_DIR to the ONNX Runtime session headers")
endif()
if(NOT EXISTS "${BF16_SOURCE_DIR}/chunked_prefill.cuh")
  message(FATAL_ERROR "Set BF16_SOURCE_DIR to the qualified BF16 CUDA snapshot")
endif()
find_package(CUDAToolkit REQUIRED)
set(CMAKE_CUDA_ARCHITECTURES 90)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
add_library(lightning_q4_ops SHARED q4_ops.cu)
target_include_directories(lightning_q4_ops PRIVATE "${ORT_INCLUDE_DIR}" "${BF16_SOURCE_DIR}")
target_link_libraries(lightning_q4_ops PRIVATE CUDA::cublas CUDA::cublasLt)
target_compile_options(lightning_q4_ops PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-O3;--fmad=false;-lineinfo>")
set_target_properties(lightning_q4_ops PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
