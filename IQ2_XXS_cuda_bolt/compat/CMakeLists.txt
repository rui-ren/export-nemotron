cmake_minimum_required(VERSION 3.22)
set(CMAKE_CUDA_ARCHITECTURES OFF)
project(bolt_upstream_q8_compat LANGUAGES CXX CUDA)
set(ORT_INCLUDE_DIR "" CACHE PATH "Pinned ONNX Runtime headers")
if(NOT EXISTS "${ORT_INCLUDE_DIR}/onnxruntime_cxx_api.h")
  message(FATAL_ERROR "ORT_INCLUDE_DIR must contain onnxruntime_cxx_api.h")
endif()
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CUDA_FLAGS_RELEASE "-O3")
set(CMAKE_CXX_FLAGS_RELEASE "-O3")
add_library(bolt_q8_fp16 SHARED q8_rank_compat.cu)
target_include_directories(bolt_q8_fp16 PRIVATE "${ORT_INCLUDE_DIR}")
target_compile_options(bolt_q8_fp16 PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-arch=sm_90;-lineinfo>")
set_target_properties(bolt_q8_fp16 PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
